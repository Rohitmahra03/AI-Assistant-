Full-Stack Groq Backend Integration Guide

Securely proxy requests to Groq Cloud using your Node.js or Python backend

🔒 Security Best Practice: Never expose GROQ_API_KEY in frontend JavaScript. Always send requests from the browser to your backend route (POST /api/chat), which interacts with Groq servers safely.

1. Express.js / Node.js Backend Route (server.js) 

import express from 'express';
import Groq from 'groq-sdk';
import dotenv from 'dotenv';
dotenv.config();
const app = express();
app.use(express.json());
const groq = new Groq({ apiKey: process.env.GROQ_API_KEY });
app.post('/api/chat', async (req, res) => {
  try {
    const { messages, subject, level, language } = req.body;
    const systemPrompt = `You are School AI, an expert, patient school teacher. 
Subject: ${subject}, Level: ${level}, Language: ${language}.
Format answers with formulas, steps, real examples, and clear conclusions.`;
    const chatCompletion = await groq.chat.completions.create({
      messages: [{ role: 'system', content: systemPrompt }, ...messages],
      model: 'llama-3.3-70b-versatile',
      temperature: 0.3,
    });
    res.json({ reply: chatCompletion.choices[0]?.message?.content || '' });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});
app.listen(3000, () => console.log('School AI backend active on port 3000'));

3. Frontend Fetch Hook 

async function askSchoolAI(question) {
  const response = await fetch('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      messages: [{ role: 'user', content: question }],
      subject: 'Physics',
      level: 'Intermediate',
      language: 'Hinglish'
    })
  });
  const data = await response.json();
  return data.reply;
}
